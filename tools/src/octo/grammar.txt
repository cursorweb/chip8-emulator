S -> decl* EOF
decl -> alias | const | label
      | stmt

alias -> ":alias" IDENT REGISTER
const -> ":const" IDENT NUM
label -> ":" IDENT
reg -> REGISTER | IDENT
num -> NUM | IDENT

stmt -> ifstmt | loopstmt | whilestmt
      | instr
      | assignment
      | NUM         # literally just insert this into the binary

instr -> ( "return" | ";" )
       | ( "clear" | "cls" )
       | "bcd" reg
       | "save" reg
       | "load" reg
       | ( "sprite" | "draw" ) reg reg num
       | "jump" num
       | "jump0" num
       | again

assignment -> "delay" ":=" reg
            | "buzzer" ":=" reg
            | "i" ":=" num         # Set i to NUM
            | "i" ":=" "hex" reg   # eg i := hex v0, v0 := 0xF sets i to font character F sprite
            | "i" "+=" reg
            | reg ":=" ( num | reg )
            | reg ":=" "random" num
            | reg ":=" "delay"
            | reg ":=" "key"
            | reg "+=" ( num | reg )
            | reg "-=" ( num | reg ) # subf: vx = vx - vy
            | reg "=-" reg           # subnf: vx = vy - vx
            | reg "|=" reg
            | reg "&=" reg
            | reg "^=" reg
            | reg ">>=" reg
            | reg "<<=" reg

ifstmt -> "if" comparison "then" # NOTE! this is really a seq
        | "if" comparison "begin" stmt* ( "else" stmt* )? "end"
loopstmt -> "loop" stmt* again # NOTE! Again is a statement that jumps back to the loop start. (Semantic + Real use)
whilestmt -> "while" comparison  # NOTE! This acts more like a break

comparison -> reg "==" ( num | reg )
            | reg "!=" ( num | reg )
            | reg "key"
            | reg "-" "key"
            | reg ( ">" | "<" | ">=" | "<=" ) ( num | reg )
